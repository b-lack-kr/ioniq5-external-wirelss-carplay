"""Authenticated 1.0.12 -> 1.0.15 upgrades retain the old transaction for rollback."""
import hashlib
import os
from pathlib import Path
import shutil
import unittest
from zipfile import ZipFile

import test_installer as fixture
import test_missing_map as missing

PUBLIC_112 = Path(os.environ.get('NEEV_PUBLIC_112_USB', str(
    fixture.PROJECT / '배포용/1.0.12/NEEV_260902_CarPlay_1.0.12_USB.zip')))
PUBLIC_112_SHA = 'f9809de66b866592539d4aa489ebb6d3e7fe7e0d00aaabaf9208582e5482f2da'
KIT_112_MD5 = '0e187a3871ec165fabf132d685f4be7a'
MOD_112_MD5 = '8b19ded5ac2b93da86a0a65eeb9fb73e'
write = fixture.write
OLD_PREFS = '{"carplay_tbt":false,"carplay_vehicle_speed_fast":false,"carplay_location_diagnostics":false,"ar_view":false,"external_dongle":true}'


def tree(folder):
    return {str(p.relative_to(folder)): ('link', os.readlink(p)) if p.is_symlink()
            else ('file', hashlib.sha256(p.read_bytes()).hexdigest()) if p.is_file()
            else ('dir',)
            for p in sorted(folder.rglob('*'))}


class Upgrade113(unittest.TestCase):
    run_usb = fixture.Installation.run_usb
    public101 = fixture.Installation.public101
    restore = fixture.Installation.restore

    def setUp(self):
        fixture.Installation.setUp(self)

    def public112(self):
        self.public101()
        self.assertEqual(hashlib.sha256(PUBLIC_112.read_bytes()).hexdigest(), PUBLIC_112_SHA)
        old = self.W/'public-1.0.12'
        with ZipFile(PUBLIC_112) as z:
            for name in z.namelist():
                if name.startswith('NEEV_CP/') and not name.endswith('/'):
                    dest = old/'kit'/name[len('NEEV_CP/'):]
                    dest.parent.mkdir(parents=True, exist_ok=True)
                    dest.write_bytes(z.read(name))
        manifest = old/'kit/MANIFEST.md5'
        self.assertEqual(fixture.md5(manifest.read_bytes()), KIT_112_MD5)
        for line in manifest.read_text().splitlines():
            digest, name = line.split()
            self.assertEqual(fixture.md5((old/'kit'/name).read_bytes()), digest)
        self.assertEqual(fixture.md5((old/'kit/payload/mod/META-HMGAPI/manifest.json').read_bytes()),MOD_112_MD5)
        write(old/'STATUS','INSTALLED\n');write(old/'mode','1.0.11\n')
        write(old/'backup/retained-prior-record','do not modify this rollback')
        shutil.rmtree(self.M);shutil.copytree(old/'kit/payload/mod',self.M)
        write(self.W/'UPGRADE_CURRENT.txt','1.0.12\n')
        write(self.root/'hmg-version','hmgapid 1.0.2')
        write(self.W/'vendor/hkg-2.0.2/asset','fixture\n')
        write(self.H/'mods/neev-carplay-mute-pause/mod.json','unrelated mute package')
        write(self.H/'settings/neev-carplay-mute-pause.json','{"enabled":false}')
        write(self.H/'settings/neev-usb-carplay.json',OLD_PREFS)
        return old

    def assert_base(self, old_record, old_mod):
        self.assertEqual(tree(self.W/'public-1.0.12'),old_record)
        self.assertEqual(tree(self.M),old_mod)
        self.assertEqual((self.W/'UPGRADE_CURRENT.txt').read_text().strip(),'1.0.12')
        self.assertEqual((self.W/'config/hotspot.conf').read_text(),'prior-secret')
        self.assertEqual((self.W/'config/ap-identity').read_text(),'prior-id')
        self.assertEqual((self.H/'settings/neev-usb-carplay.json').read_text(),OLD_PREFS)
        self.assertEqual((self.H/'mods/neev-carplay-mute-pause/mod.json').read_text(),'unrelated mute package')
        self.assertEqual((self.H/'settings/neev-carplay-mute-pause.json').read_text(),'{"enabled":false}')
        self.assertFalse((self.M/'.disabled').exists())

    def test_upgrade_repeat_and_restore_to_exact_112_preserve_records_settings_and_mute(self):
        old=self.public112();old_record=tree(old);old_mod=tree(self.M)
        self.assertEqual(self.run_usb(),'COMPLETE')
        self.assertEqual((self.W/'UPGRADE_CURRENT.txt').read_text().strip(),'1.0.15')
        self.assertEqual((self.R/'mode').read_text().strip(),'1.0.12')
        self.assertEqual(tree(old),old_record)
        self.assertEqual((self.W/'config/hotspot.conf').read_text(),'prior-secret')
        self.assertEqual((self.W/'config/ap-identity').read_text(),'prior-id')
        self.assertEqual((self.H/'settings/neev-usb-carplay.json').read_text(),OLD_PREFS)
        self.assertEqual((self.H/'mods/neev-carplay-mute-pause/mod.json').read_text(),'unrelated mute package')
        self.assertEqual((self.H/'settings/neev-carplay-mute-pause.json').read_text(),'{"enabled":false}')
        self.assertEqual(self.run_usb(),'COMPLETE')
        self.restore();self.assert_base(old_record,old_mod)
        # Enabled independent consumers intentionally retain their shared map.
        self.assertTrue(self.map.is_dir())
        (self.usb/'NEEV-CP-RESTORE').unlink();self.assertEqual(self.run_usb(),'COMPLETE')
        self.assertEqual((self.W/'UPGRADE_CURRENT.txt').read_text().strip(),'1.0.15')

    def test_enabled_independent_mute_is_unchanged_on_upgrade_and_rollback(self):
        self.public112()
        addon=self.H/'mods/neev-carplay-mute-pause'
        settings=self.H/'settings/neev-carplay-mute-pause.json'
        write(addon/'hooks.jar','independent signed payload')
        write(settings,'{"enabled":true,"pause_on_mute":true}')
        before=tree(addon);prefs=settings.read_bytes()
        self.assertEqual(self.run_usb(),'COMPLETE')
        self.assertEqual(tree(addon),before);self.assertEqual(settings.read_bytes(),prefs)
        self.restore()
        self.assertEqual(tree(addon),before);self.assertEqual(settings.read_bytes(),prefs)
        self.assertTrue(self.map.is_dir())

                         '0eb75dc5798888669cc44df2edc034a5ea488557e4829253472b3e05925ba373')
        with ZipFile(release109) as z:

                if name.startswith('NEEV_CP/') and not name.endswith('/'):
                    dest = old109/'kit'/name[len('NEEV_CP/'):]
                    dest.parent.mkdir(parents=True, exist_ok=True)
                    dest.write_bytes(z.read(name))
        write(old109/'STATUS', 'INSTALLED\n'); write(old109/'mode', '1.0.7\n')
        old110 = self.W/'public-1.0.10'
        release110 = fixture.PROJECT/'배포용/1.0.10/NEEV_260902_CarPlay_1.0.10_USB.zip'
        self.assertEqual(hashlib.sha256(release110.read_bytes()).hexdigest(),
                         '6b1d0798faa69601cbd264626238c2c100d4335c6f838128b4a81f0e31bad160')
        with ZipFile(release110) as z:
            for name in z.namelist():
                if name.startswith('NEEV_CP/') and not name.endswith('/'):
                    dest = old110/'kit'/name[len('NEEV_CP/'):]
                    dest.parent.mkdir(parents=True, exist_ok=True)
                    dest.write_bytes(z.read(name))
        write(old110/'STATUS', 'INSTALLED\n'); write(old110/'mode', '1.0.9\n')
        old111 = self.W/'public-1.0.11'
        release111 = fixture.PROJECT/'배포용/1.0.11/NEEV_260902_CarPlay_1.0.11_USB.zip'
        self.assertEqual(hashlib.sha256(release111.read_bytes()).hexdigest(),
                         '34f4f5037c5e622c41a03b61ac8d807a805d5099349239c0a738781aef1f3ca4')

    def test_manual_uninstall_112_then_clean_113_retains_authenticated_archive(self):
        old=self.public112();old_record=tree(old)
        shutil.rmtree(self.M);(self.H/'state/applied.fixture').unlink()
        write(self.usb/'packages/usbwifi.hotspot.zip','vendor-fixture')
        self.assertEqual(self.run_usb(),'COMPLETE')
        self.assertEqual((self.R/'mode').read_text().strip(),'fresh')
        archives=list(self.W.parent.glob('neev-usbcp-removed-*'))
        self.assertEqual(len(archives),1)
        self.assertEqual(tree(archives[0]/'public-1.0.12'),old_record)
        self.assertEqual((archives[0]/'.manual-uninstall-settings.json').read_text(),OLD_PREFS)
        self.assertFalse((self.H/'settings/neev-usb-carplay.json').exists())
        self.assertEqual((archives[0]/'config/ap-identity').read_text(),'prior-id')
        self.assertEqual((self.H/'settings/neev-carplay-mute-pause.json').read_text(),'{"enabled":false}')

    def test_damaged_removed_112_record_is_not_archived_or_replaced(self):
        old=self.public112()
        shutil.rmtree(self.M);(self.H/'state/applied.fixture').unlink()
        write(self.usb/'packages/usbwifi.hotspot.zip','vendor-fixture')
        write(old/'kit/operate.sh','damaged previous kit')
        before=tree(self.W);hmg=tree(self.H)
        self.assertEqual(self.run_usb(False),'UNOWNED_REMOVED_SETUP')
        self.assertEqual(tree(self.W),before);self.assertEqual(tree(self.H),hmg)
        self.assertFalse(list(self.W.parent.glob('neev-usbcp-removed-*')))

    def test_removed_completed_112_upgrade_retains_both_authenticated_records(self):
        old=self.public112();self.run_usb()
        old_record=tree(old);current_record=tree(self.R)
        shutil.rmtree(self.M);(self.H/'state/applied.fixture').unlink()
        write(self.usb/'packages/usbwifi.hotspot.zip','vendor-fixture')
        self.assertEqual(self.run_usb(),'COMPLETE')
        self.assertEqual((self.R/'mode').read_text().strip(),'fresh')
        archives=list(self.W.parent.glob('neev-usbcp-removed-*'))
        self.assertEqual(len(archives),1)
        self.assertEqual(tree(archives[0]/'public-1.0.12'),old_record)
        self.assertEqual(tree(archives[0]/'public-1.0.15'),current_record)

    def test_failed_install_restores_exact_112_map_receipt_and_prior_record(self):
        old=self.public112();old_record=tree(old);old_mod=tree(self.M)
        write(self.root/'fault','engine-install')
        self.assertEqual(self.run_usb(False),'INSTALL_FAILED_RESTORED')
        self.assert_base(old_record,old_mod)
        self.assertEqual((self.map/'car.json').read_text(),'previous-map')
        self.assertEqual((self.H/'state/map-check.json').read_text(),'previous-map-check')
        self.assertEqual((self.R/'STATUS').read_text().strip(),'RESTORED')
        self.assertEqual(self.run_usb(),'COMPLETE')

    def test_failed_map_verification_restores_112_without_mod_install(self):

        self.assertEqual(self.run_usb(False),'INSTALL_FAILED_RESTORED')

        self.assertEqual((self.map/'car.json').read_text(),'previous-map')
        self.assertEqual((self.H/'state/map-check.json').read_text(),'previous-map-check')

    def test_interrupted_mod_install_rolls_back_after_reboot_and_can_retry(self):
        old=self.public112();old_record=tree(old);old_mod=tree(self.M)
        write(self.root/'fault','engine-install-crash')
        self.run_usb(False)
        self.assertEqual((self.R/'STATUS').read_text().strip(),'INSTALLING')
        write(self.root/'proc/sys/kernel/random/boot_id','cdef-cdef')
        self.assertEqual(self.run_usb(),'COMPLETE')
        self.assert_base(old_record,old_mod)
        self.assertEqual((self.R/'STATUS').read_text().strip(),'RESTORED')
        self.assertEqual(self.run_usb(),'COMPLETE')

    def test_unknown_or_damaged_112_kit_is_rejected_before_staging(self):
        old=self.public112();write(old/'kit/operate.sh','changed old transaction')
        state=tree(self.H)
        self.assertEqual(self.run_usb(False),'UNOWNED_PUBLIC_112')
        self.assertFalse(self.R.exists());self.assertEqual(tree(self.H),state)

    def test_tampered_installed_112_is_rejected_before_live_changes(self):
        self.public112();write(self.M/'hooks.jar','tampered')
        state=tree(self.H)
        self.assertEqual(self.run_usb(False),'EXISTING_MOD_SIGNATURE')
        self.assertEqual(tree(self.H),state)
        self.assertEqual((self.W/'UPGRADE_CURRENT.txt').read_text().strip(),'1.0.12')
        self.assertEqual((self.R/'STATUS').read_text().strip(),'PREPARED')

    def test_unfinished_or_disabled_112_is_rejected(self):
        old=self.public112()
        for status in ('PREPARED','INSTALLING','RESTORING','RESTORED','unknown'):
            with self.subTest(status=status):
                write(old/'STATUS',status)
                self.assertEqual(self.run_usb(False),'UNOWNED_PUBLIC_112')
                self.assertFalse(self.R.exists())
        write(old/'STATUS','INSTALLED');write(self.M/'.disabled','')
        self.assertEqual(self.run_usb(False),'EXISTING_MOD_DISABLED')
        self.assertFalse(self.R.exists())

    def test_unrecognized_112_origin_is_rejected_before_live_changes(self):
        old=self.public112()
        write(old/'mode','1.0.8')
        before=tree(self.H)
        self.assertEqual(self.run_usb(False),'UNOWNED_PUBLIC_112')
        self.assertEqual(tree(self.H),before);self.assertFalse(self.R.exists())

    def test_old_prepared_kit_under_new_release_path_is_not_refreshed_or_executed(self):
        old=self.public112()

        state=tree(self.W)

    def test_manual_removal_after_112_rollback_can_clean_install_without_losing_archive(self):

    def test_existing_new_transaction_tamper_prevents_install_and_rollback(self):
        self.public112();self.run_usb();write(self.R/'kit/operate.sh','foreign')
        current=tree(self.H)
        self.assertEqual(self.run_usb(False),'LOCAL_TRANSACTION_DAMAGED')
        write(self.usb/'NEEV-CP-RESTORE','')
        self.assertEqual(self.run_usb(False),'LOCAL_TRANSACTION_DAMAGED')
        self.assertEqual(tree(self.H),current)

    def test_missing_map_112_recovery_uses_real_verification(self):
        old=self.public112();shutil.rmtree(self.map)
        mock=(self.bin/'app_process').read_text().replace(
            " elif op=='map':print('hook map: probed for Ioniq 5, hooks on')",
            " elif op=='map':\n  print('hook map: probed for Ioniq 5, hooks on' if (H/'maps/hyundai/neev/kor/260902').exists() else "+repr(missing.NO_MAP)+")")
        write(self.bin/'app_process',mock,True)
